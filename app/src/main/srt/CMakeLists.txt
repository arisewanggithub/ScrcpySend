cmake_minimum_required(VERSION 3.22.1)
project(screencopy_srt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(THIRD_PARTY "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party" ABSOLUTE)
set(SRT_ARCHIVE "${THIRD_PARTY}/srt-1.5.7.tar.gz")
file(SHA256 "${SRT_ARCHIVE}" ACTUAL)
file(READ "${THIRD_PARTY}/srt-1.5.7.sha256" EXPECTED)
string(STRIP "${EXPECTED}" EXPECTED)
if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Pinned libsrt source checksum mismatch")
endif()
set(SRT_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/sources/srt-1.5.7")
if(NOT EXISTS "${SRT_SOURCE}/.verified-v2-${ACTUAL}")
    file(ARCHIVE_EXTRACT INPUT "${SRT_ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/sources"
        PATTERNS "srt-1.5.7/CMakeLists.txt" "srt-1.5.7/*.in" "srt-1.5.7/srtcore/*"
        "srt-1.5.7/common/*" "srt-1.5.7/scripts/*" "srt-1.5.7/haicrypt/*"
        "srt-1.5.7/LICENSE" "srt-1.5.7/*.cmake" "srt-1.5.7/cmake_object_lib_support.c")
    file(WRITE "${SRT_SOURCE}/.verified-v2-${ACTUAL}" "${ACTUAL}")
endif()
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(ENABLE_APPS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_UNITTESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_ENCRYPTION OFF CACHE BOOL "" FORCE)
set(ENABLE_CXX11 ON CACHE BOOL "" FORCE)
add_subdirectory("${SRT_SOURCE}" srt-build EXCLUDE_FROM_ALL)
add_library(screencopy_srt SHARED srt_transport.cpp)
target_include_directories(screencopy_srt PRIVATE
    "${SRT_SOURCE}/srtcore" "${SRT_SOURCE}/common" "${CMAKE_CURRENT_BINARY_DIR}/srt-build")
target_link_libraries(screencopy_srt PRIVATE srt_static log)
target_compile_options(screencopy_srt PRIVATE -Wall -Wextra -Werror)
target_link_options(screencopy_srt PRIVATE "-Wl,-z,max-page-size=16384")
